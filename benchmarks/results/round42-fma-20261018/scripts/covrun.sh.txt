#!/bin/sh
# covrun.sh TAG CORES: both packages' suites with coverage, then a marker.
cd ~/r42 || exit 1
sha256sum cov-fft.test cov-k.test > cov-$1.sha
GOMAXPROCS=2 taskset -c "$2" ./cov-fft.test -test.timeout=2h -test.coverprofile=cov-fft-$1.out > suite-fft-$1.log 2>&1
GOMAXPROCS=2 taskset -c "$2" ./cov-k.test -test.timeout=1h -test.coverprofile=cov-k-$1.out > suite-k-$1.log 2>&1
echo DONE > cov-$1.done
