#!/bin/sh
# r32calm.sh THRESH MAXSEC cmd... : wait until the 1-minute load is below
# THRESH (checked every 15 s, at most MAXSEC), then run cmd.
th=$1; max=$2; shift 2
t=0
while :; do
  l=$(cut -d' ' -f1 /proc/loadavg)
  if awk -v l="$l" -v t="$th" 'BEGIN{exit !(l<t)}'; then break; fi
  if [ "$t" -ge "$max" ]; then echo "R32CALM timeout load=$l after ${t}s"; exit 3; fi
  sleep 15; t=$((t+15))
done
echo "R32CALM go after ${t}s: $(cat /proc/loadavg)"
exec "$@"
