#!/bin/sh
# r32suite.sh DIR: both packages' suites with coverage (fftcov.test, kerncov.test
# built with -coverpkg of both), then the merged total.
D=$1
mkdir -p $D && cd $D && tar xzf ../testdata.tgz 2>/dev/null
echo "# $(hostname) $(date +%T) load $(cut -d' ' -f1-3 /proc/loadavg)"
../fftcov.test -test.count=1 -test.coverprofile=fft.cov > fft.log 2>&1; echo "fft exit $?"; tail -2 fft.log
(cd ../ && mkdir -p kdir) ; ../kerncov.test -test.count=1 -test.coverprofile=kern.cov > kern.log 2>&1; echo "kernels exit $?"; tail -2 kern.log
echo "# end $(date +%T) load $(cut -d' ' -f1-3 /proc/loadavg)"
