// an: scores factorization rules against a TestZmSweep file: for each rule,
// over the lengths whose rule choice was timed, the geometric mean and the
// worst of best/rule, and the geometric mean and worst of rule/current.
package main

import (
	"bufio"
	"fmt"
	"math"
	"os"
	"sort"
	"strconv"
	"strings"
)

type cand struct {
	f string
	t float64
}

func r8max(e int) []int {
	n8 := e / 3
	var f []int
	switch e % 3 {
	case 1:
		if e >= 4 {
			n8--
			f = []int{4, 4}
		} else {
			f = []int{2}
		}
	case 2:
		f = []int{4}
	}
	for range n8 {
		f = append([]int{8}, f...)
	}
	return f
}

func rev(f []int) []int {
	g := make([]int, len(f))
	for i := range f {
		g[len(f)-1-i] = f[i]
	}
	return g
}

func name(f []int) string {
	s := make([]string, len(f))
	for i, r := range f {
		s[i] = strconv.Itoa(r)
	}
	return strings.Join(s, "x")
}

func cnt(n int) (e, a, b int) {
	for n%2 == 0 {
		n /= 2
		e++
	}
	for n%3 == 0 {
		n /= 3
		a++
	}
	for n%5 == 0 {
		n /= 5
		b++
	}
	return
}

func rep(f []int, r, c int) []int {
	for range c {
		f = append(f, r)
	}
	return f
}

// chain: 3s, 5s, then the power-of-two tail (radix 4 before 8).
func chain(n int) string {
	e, a, b := cnt(n)
	return name(append(rep(rep(nil, 3, a), 5, b), rev(r8max(e))...))
}

// chain5: 5s first.
func chain5(n int) string {
	e, a, b := cnt(n)
	return name(append(rep(rep(nil, 5, b), 3, a), rev(r8max(e))...))
}

// pfaLast: a final radix-20 (b >= 1, e >= 2) else radix-12 (a >= 1, e >= 2) pass, the rest a chain.
func pfaLast(n int, pref20 bool) string {
	e, a, b := cnt(n)
	if e >= 2 && (b >= 1 && (pref20 || a == 0)) {
		return name(append(append(rep(rep(nil, 3, a), 5, b-1), rev(r8max(e-2))...), 20))
	}
	if e >= 2 && a >= 1 {
		return name(append(append(rep(rep(nil, 3, a-1), 5, b), rev(r8max(e-2))...), 12))
	}
	return chain(n)
}

func main() {
	f, _ := os.Open(os.Args[1])
	sc := bufio.NewScanner(f)
	sc.Buffer(make([]byte, 1<<20), 1<<20)
	rows := map[int][]cand{}
	for sc.Scan() {
		p := strings.Fields(sc.Text())
		if len(p) < 3 || p[0] != "SWEEP" {
			continue
		}
		n, _ := strconv.Atoi(p[1])
		for _, x := range p[2:] {
			fv := strings.SplitN(x, "=", 2)
			tr := strings.Split(fv[1], "/")
			t, _ := strconv.ParseFloat(tr[0], 64)
			rows[n] = append(rows[n], cand{fv[0], t})
		}
	}
	var ns []int
	for n := range rows {
		ns = append(ns, n)
	}
	sort.Ints(ns)
	rules := map[string]func(int) string{
		"cur":     func(n int) string { return rows[n][0].f },
		"keepLast": func(n int) string { return keepLast(rows[n][0].f, n) },
		"chain":   chain,
		"chain5":  chain5,
		"pfa20":   func(n int) string { return pfaLast(n, true) },
		"pfa12":   func(n int) string { return pfaLast(n, false) },
		"bestOfChainPfa": func(n int) string {
			best, bt := "", math.Inf(1)
			for _, s := range []string{chain(n), chain5(n), pfaLast(n, true), pfaLast(n, false)} {
				for _, c := range rows[n] {
					if c.f == s && c.t < bt {
						best, bt = s, c.t
					}
				}
			}
			return best
		},
	}
	var names []string
	for k := range rules {
		names = append(names, k)
	}
	sort.Strings(names)
	verbose := len(os.Args) > 2
	for _, k := range names {
		var lb, lc float64
		wb, wc := 0.0, math.Inf(1)
		m, miss := 0, 0
		wbn, wcn := 0, 0
		for _, n := range ns {
			c := rows[n]
			bt := math.Inf(1)
			for _, x := range c {
				bt = min(bt, x.t)
			}
			want := rules[k](n)
			rt := -1.0
			for _, x := range c {
				if x.f == want {
					rt = x.t
				}
			}
			if rt < 0 {
				miss++
				if verbose && k == os.Args[2] {
					fmt.Printf("  %d %s not timed\n", n, want)
				}
				continue
			}
			m++
			lb += math.Log(rt / bt)
			lc += math.Log(c[0].t / rt)
			if rt/bt > wb {
				wb, wbn = rt/bt, n
			}
			if c[0].t/rt < wc {
				wc, wcn = c[0].t/rt, n
			}
			if verbose && k == os.Args[2] {
				fmt.Printf("  %6d %-20s rule/best %.3f cur/rule %.3f\n", n, want, rt/bt, c[0].t/rt)
			}
		}
		fmt.Printf("%-15s timed %3d missing %3d  rule÷best geo %.4f worst %.3f (%d)  cur÷rule geo %.4f worst %.3f (%d)\n",
			k, m, miss, math.Exp(lb/float64(m)), wb, wbn, math.Exp(lc/float64(m)), wc, wcn)
	}
}
package main

import (
	"strconv"
	"strings"
)

// keepLast: the current factorization with every PFA pass but the last one
// broken into its factors (12 = 3·4, 20 = 5·4, 10 = 5·2, 15 = 3·5): 3s,
// then 5s, then the power of two as radix 4 before 8, then the last pass.
// If the last pass is 10 or 15, it is broken too (no PFA at all).
func keepLast(cur string, n int) string {
	var f []int
	for _, s := range strings.Split(cur, "x") {
		r, _ := strconv.Atoi(s)
		f = append(f, r)
	}
	last := f[len(f)-1]
	pfa := last == 12 || last == 20
	m := n
	if pfa {
		m /= last
	}
	e, a, b := cnt(m)
	g := append(rep(rep(nil, 3, a), 5, b), rev(r8max(e))...)
	if pfa {
		g = append(g, last)
	}
	return name(g)
}
