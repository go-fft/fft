//go:build amd64 || arm64

package fft

import (
	"fmt"
	"testing"
)

// BenchmarkR22Order times Plan32's Stockham engine with the kernels on, in the
// pocketfft radix order (pf) and the odd-first order (odd), per length.
func BenchmarkR22Order(b *testing.B) {
	for _, n := range []int{64, 256, 512, 1000, 1024, 1080, 1296, 1920, 2048, 4096, 8192, 20160, 65536, 1 << 20} {
		x := make([]complex64, n)
		for i, v := range cmplxSignal(n) {
			x[i] = complex64(v)
		}
		dst := make([]complex64, n)
		for _, odd := range []bool{false, true} {
			name := "pf"
			if odd {
				name = "odd"
			}
			p := newSKPlan32Factors(n, skFactorizeOrder(n, odd))
			b.Run(fmt.Sprintf("%d/%s", n, name), func(b *testing.B) {
				for b.Loop() {
					p.transform(dst, x, false)
				}
			})
		}
	}
}
